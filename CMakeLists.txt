cmake_minimum_required(VERSION 3.18)
project(carma_pack_modern LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Python REQUIRED COMPONENTS Interpreter Development.Module)
find_package(pybind11 CONFIG REQUIRED)
find_package(Armadillo REQUIRED)
find_package(Boost CONFIG REQUIRED COMPONENTS filesystem timer)
set(CORE src/carmcmc.cpp src/carpack.cpp src/kfilter.cpp src/proposals.cpp src/samplers.cpp src/random.cpp src/steps.cpp)
add_library(carma_core STATIC ${CORE})
set_target_properties(carma_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(carma_core PUBLIC src/include ${ARMADILLO_INCLUDE_DIRS})
target_link_libraries(carma_core PUBLIC ${ARMADILLO_LIBRARIES} Boost::filesystem Boost::timer)
pybind11_add_module(_carmcmc NO_EXTRAS src/pybind_wrapper.cpp)
target_link_libraries(_carmcmc PRIVATE carma_core)
install(TARGETS _carmcmc LIBRARY DESTINATION carmcmc)
add_executable(reference_driver modern_tests/reference_driver.cpp)
target_link_libraries(reference_driver PRIVATE carma_core)
add_executable(initialization_probe EXCLUDE_FROM_ALL modern_tests/initialization_probe.cpp)
target_link_libraries(initialization_probe PRIVATE carma_core)
