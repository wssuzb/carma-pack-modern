cmake_minimum_required(VERSION 3.18)
project(upstream_oracle LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
find_package(Armadillo REQUIRED)
find_package(Boost CONFIG REQUIRED COMPONENTS filesystem timer)
add_library(upstream_core STATIC carmcmc.cpp carpack.cpp kfilter.cpp proposals.cpp samplers.cpp random.cpp steps.cpp)
target_include_directories(upstream_core PUBLIC include ${ARMADILLO_INCLUDE_DIRS})
target_link_libraries(upstream_core PUBLIC ${ARMADILLO_LIBRARIES} Boost::filesystem Boost::timer)
add_executable(upstream_initialization_probe ../initialization_probe.cpp)
target_link_libraries(upstream_initialization_probe PRIVATE upstream_core)
add_executable(upstream_reference_driver ../reference_driver.cpp)
target_link_libraries(upstream_reference_driver PRIVATE upstream_core)
